sequenceDiagram
    title Flow 8 - Logout
    actor User
    participant API as WebApi Controllers
    participant Med as MediatR
    participant H as Application Handlers
    participant Repo as Repository/UoW
    participant DB as PostgreSQL

    User->>API: 1: Request Logout
    activate API
    API->>Med: 1.1: Redirect Send(Command)
    activate Med
    Med->>H: 1.1.1: Dispatch command
    activate H
    H->>Repo: 1.1.1.1: Load refresh token/session
    activate Repo
    Repo->>DB: 1.1.1.1.1: Select refresh token by user/session
    activate DB
    DB-->>Repo: 1.1.1.1.2: Token result
    deactivate DB
    Repo-->>H: 1.1.1.2: Repository result
    deactivate Repo
    H->>Repo: 1.1.1.3: Revoke refresh token/session
    activate Repo
    Repo->>DB: 1.1.1.3.1: Update refresh token status + Commit
    activate DB
    DB-->>Repo: 1.1.1.3.2: DB result
    deactivate DB
    Repo-->>H: 1.1.1.4: Repository result
    deactivate Repo

    alt 1.2: Logout success
        H-->>Med: 1.2.1: Return Success(Logout)
        Med-->>API: 1.2.2: Return success response
        API-->>User: 1.2.3: Return Logout success
        User-->>User: 1.2.4: Show success message
    else 1.3: Logout failed
        H-->>Med: 1.3.1: Return Failure(Logout)
        Med-->>API: 1.3.2: Return failure response
        API-->>User: 1.3.3: Return Logout failure
        User-->>User: 1.3.4: Show error message
    end

    deactivate H
    deactivate Med
    deactivate API
