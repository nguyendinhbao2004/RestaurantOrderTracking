sequenceDiagram
    title Flow 8 - Logout
    actor User
    participant UI as Web/Mobile UI
    participant Controller as WebApi Controller
    participant Handler as Feature Handler
    participant Repo as Repository/UoW
    participant DB as PostgreSQL

    User->>UI: 1: Tap Logout
    activate UI
    UI->>Controller: 1.1: HTTP POST /api/Auth/logout
    activate Controller
    Controller->>Handler: 1.1.1: Invoke LogoutHandler
    activate Handler
    Handler->>Repo: 1.1.1.1: Load refresh token/session
    activate Repo
    Repo->>DB: 1.1.1.1.1: Select refresh token by user/session
    activate DB
    DB-->>Repo: 1.1.1.1.2: Token result
    deactivate DB
    Repo-->>Handler: 1.1.1.1.3: Repository result
    deactivate Repo
    Handler->>Repo: 1.1.1.1.4: Revoke refresh token/session
    activate Repo
    Repo->>DB: 1.1.1.1.4.1: Update refresh token status + Commit
    activate DB
    DB-->>Repo: 1.1.1.1.4.2: DB result
    deactivate DB
    Repo-->>Handler: 1.1.1.1.4.3: Repository result
    deactivate Repo

    alt 1.2: Logout success
        Handler-->>Controller: 1.2.1: Return Success(Logout)
        Controller-->>UI: 1.2.2: Return Logout success
        UI-->>User: 1.2.3: Show success message
    else 1.3: Logout failed
        Handler-->>Controller: 1.3.1: Return Failure(Logout)
        Controller-->>UI: 1.3.2: Return Logout failure
        UI-->>User: 1.3.3: Show error message
    end

    deactivate Handler
    deactivate Controller
    deactivate UI
