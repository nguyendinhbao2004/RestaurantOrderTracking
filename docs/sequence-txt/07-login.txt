sequenceDiagram
    title Flow 7 - Login
    actor User
    participant UI as Web/Mobile UI
    participant Controller as WebApi Controller
    participant Handler as Feature Handler
    participant Repo as Repository/UoW
    participant DB as PostgreSQL

    User->>UI: 1: Submit credentials
    activate UI
    UI->>Controller: 1.1: HTTP POST /api/Auth/login
    activate Controller
    Controller->>Handler: 1.1.1: Invoke LoginHandler
    activate Handler
    Handler->>Handler: 1.1.1.1: Validate login input

    alt 1.2: Invalid email format
        Handler-->>Controller: 1.2.1: Return Failure(Validation)
        Controller-->>UI: 1.2.2: Return Login failure
        UI-->>User: 1.2.3: Show error message
    else 1.3: Input valid
        Handler->>Repo: 1.3.1.1: Load account by email
        activate Repo
        Repo->>DB: 1.3.1.1.1: Select account
        activate DB
        DB-->>Repo: 1.3.1.1.2: Account result
        deactivate DB
        Repo-->>Handler: 1.3.1.3: Repository result
        deactivate Repo
        Handler->>Handler: 1.3.1.4: Verify password

        alt 1.4: Credentials valid
            Handler->>Repo: 1.4.1.1: Save refresh token
            activate Repo
            Repo->>DB: 1.4.1.1.1: Insert refresh token + Commit
            activate DB
            DB-->>Repo: 1.4.1.1.2: DB result
            deactivate DB
            Repo-->>Handler: 1.4.1.3: Repository result
            deactivate Repo
            Handler-->>Controller: 1.4.2: Return Success(Auth)
            Controller-->>UI: 1.4.3: Return Login success
            UI-->>User: 1.4.4: Show success message
            User-->>User: 1.4.5: Navigate dashboard
        else 1.5: Wrong email or password
            Handler-->>Controller: 1.5.1: Return Failure(Auth)
            Controller-->>UI: 1.5.2: Return Login failure
            UI-->>User: 1.5.3: Show error message
        end
    end

    deactivate Handler
    deactivate Controller
    deactivate UI
